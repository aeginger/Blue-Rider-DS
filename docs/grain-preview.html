Usage

<!-- Basic usage -->
<div class="br-grain">Your content</div>

<!-- With intensity -->
<section class="br-grain br-grain--subtle">...</section>

<!-- For light backgrounds -->
<div class="br-grain br-grain--light">...</div>

<!-- Combine modifiers -->
<div class="br-grain br-grain--fine br-grain--strong">...</div>

<!-- Full page grain -->
<body class="br-page-grain">...</body>
